    score += (10 if case.relationship == 'Family' else 5)  # Relationship urgency

def suggest_best_lupon(case):
    case_type = case.case_type
    # Find member with highest success rate for this case type
    stats = Hearing.objects.filter(
        case__case_type=case_type,
        hearing_status='completed'
    ).values('lupon_member').annotate(count=Count('id'))
    return stats.order_by('-count').first()
